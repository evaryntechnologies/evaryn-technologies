"use server";

import { revalidatePath } from "next/cache";
import { requireEmployee } from "@/lib/auth/require-employee";
import { prisma } from "@/lib/db/prisma";

type Result = {
  success: boolean;
  message: string;
};

async function requireActiveAdmin() {
  const user = await requireEmployee(["ADMIN", "SUPER_ADMIN"]);

  const profile = await prisma.employeeProfile.findUnique({
    where: { authUserId: user.id },
    select: { status: true, role: true },
  });

  if (
    !profile ||
    profile.status !== "ACTIVE" ||
    !["ADMIN", "SUPER_ADMIN"].includes(profile.role)
  ) {
    throw new Error("Administrator access is not active.");
  }

  return user;
}

export async function createProject(
  formData: FormData
): Promise<Result> {
  await requireActiveAdmin();

  const title = String(formData.get("title") ?? "").trim();
  const description = String(
    formData.get("description") ?? ""
  ).trim();

  if (
    title.length < 3 ||
    title.length > 160 ||
    description.length < 10 ||
    description.length > 10000
  ) {
    return {
      success: false,
      message: "Enter a valid project title and description.",
    };
  }

  await prisma.workProject.create({
    data: {
      title,
      description,
      status: "ACTIVE",
    },
  });

  revalidatePath("/portal/admin/task-management");

  return {
    success: true,
    message: "Project created successfully.",
  };
}

export async function assignTask(
  formData: FormData
): Promise<Result> {
  const administrator = await requireActiveAdmin();

  const projectId = String(formData.get("projectId") ?? "");
  const assignedToId = String(formData.get("assignedToId") ?? "");
  const title = String(formData.get("title") ?? "").trim();
  const description = String(
    formData.get("description") ?? ""
  ).trim();

  const priority = String(formData.get("priority") ?? "NORMAL");
  const deadlineValue = String(formData.get("deadline") ?? "");

  const uuidPattern =
    /^[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i;

  if (
    !uuidPattern.test(projectId) ||
    !uuidPattern.test(assignedToId)
  ) {
    return { success: false, message: "Invalid selection." };
  }

  if (
    title.length < 3 ||
    title.length > 160 ||
    description.length < 10 ||
    description.length > 10000
  ) {
    return {
      success: false,
      message: "Enter a valid task title and description.",
    };
  }

  if (!["LOW", "NORMAL", "HIGH", "URGENT"].includes(priority)) {
    return { success: false, message: "Invalid priority." };
  }

  const deadline = deadlineValue
    ? new Date(deadlineValue + "T23:59:59+03:00")
    : null;

  if (deadlineValue && (!/^\d{4}-\d{2}-\d{2}$/.test(deadlineValue) || !deadline || Number.isNaN(deadline.getTime()))) {
    return { success: false, message: "Invalid deadline." };
  }

  const [creator, assignee, project] = await Promise.all([
    prisma.employeeProfile.findUnique({
      where: { authUserId: administrator.id },
      select: { id: true },
    }),
    prisma.employeeProfile.findUnique({
      where: { id: assignedToId },
      select: { status: true, role: true },
    }),
    prisma.workProject.findUnique({
      where: { id: projectId },
      select: { status: true },
    }),
  ]);

  if (
    !creator ||
    !assignee ||
    assignee.status !== "ACTIVE" ||
    assignee.role !== "TASKER" ||
    !project ||
    project.status !== "ACTIVE"
  ) {
    return {
      success: false,
      message: "The selected project or employee is not eligible.",
    };
  }

  await prisma.workTask.create({
    data: {
      projectId,
      assignedToId,
      createdById: creator.id,
      title,
      description,
      priority,
      status: "ASSIGNED",
      deadline,
    },
  });

  revalidatePath("/portal/admin/task-management");
  revalidatePath("/portal/tasker");

  return {
    success: true,
    message: "Task assigned successfully.",
  };
}

"use server";

import { redirect } from "next/navigation";
import { revalidatePath } from "next/cache";
import { requireEmployee } from "@/lib/auth/require-employee";
import { prisma } from "@/lib/db/prisma";

export async function reviewSubmission(formData: FormData): Promise<void> {
  const user = await requireEmployee(["REVIEWER"]);

  const submissionId = String(formData.get("submissionId") ?? "");
  const decision = String(formData.get("decision") ?? "");
  const feedback = String(formData.get("feedback") ?? "").trim();

  const uuid = /^[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i;

  if (
    !uuid.test(submissionId) ||
    !["APPROVED", "REWORK_REQUIRED"].includes(decision) ||
    feedback.length > 5000 ||
    (decision === "REWORK_REQUIRED" && feedback.length < 10)
  ) {
    redirect("/portal/reviewer/queue?error=invalid");
  }

  const reviewer = await prisma.employeeProfile.findUnique({
    where: { authUserId: user.id },
    select: { id: true, role: true, status: true },
  });

  if (!reviewer || reviewer.role !== "REVIEWER" || reviewer.status !== "ACTIVE") {
    redirect("/portal/unauthorized");
  }

  const outcome = await prisma.$transaction(async (tx) => {
    const submission = await tx.taskSubmission.findUnique({
      where: { id: submissionId },
      select: {
        id: true,
        status: true,
        employeeId: true,
        taskId: true,
        task: { select: { status: true } },
      },
    });

    if (
      !submission ||
      submission.status !== "PENDING_REVIEW" ||
      submission.task.status !== "PENDING_REVIEW" ||
      submission.employeeId === reviewer.id
    ) {
      return false;
    }

    const claimed = await tx.taskSubmission.updateMany({
      where: { id: submissionId, status: "PENDING_REVIEW" },
      data: { status: "REVIEW_IN_PROGRESS" },
    });

    if (claimed.count !== 1) return false;

    await tx.taskReview.create({
      data: {
        submissionId,
        reviewerId: reviewer.id,
        stage: "REVIEWER",
        decision,
        feedback: feedback || null,
      },
    });

    await tx.taskSubmission.update({
      where: { id: submissionId },
      data: {
        status: decision === "APPROVED" ? "PENDING_QA" : "REWORK_REQUIRED",
      },
    });

    await tx.workTask.update({
      where: { id: submission.taskId },
      data: {
        status: decision === "APPROVED" ? "PENDING_QA" : "REWORK_REQUIRED",
      },
    });

    return true;
  });

  if (!outcome) {
    redirect("/portal/reviewer/queue?error=processed");
  }

  revalidatePath("/portal/reviewer/queue");
  revalidatePath("/portal/tasker");
  revalidatePath("/portal/qa");

  redirect("/portal/reviewer/queue?result=saved");
}

"use server";

import { redirect } from "next/navigation";
import { revalidatePath } from "next/cache";
import { requireEmployee } from "@/lib/auth/require-employee";
import { prisma } from "@/lib/db/prisma";

export async function qaDecision(formData: FormData): Promise<void> {
  const user = await requireEmployee(["QA"]);

  const submissionId = String(formData.get("submissionId") ?? "");
  const decision = String(formData.get("decision") ?? "");
  const feedback = String(formData.get("feedback") ?? "").trim();

  const uuid = /^[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i;

  if (
    !uuid.test(submissionId) ||
    !["APPROVED", "REWORK_REQUIRED"].includes(decision) ||
    feedback.length > 5000 ||
    (decision === "REWORK_REQUIRED" && feedback.length < 10)
  ) {
    redirect("/portal/qa/queue?error=invalid");
  }

  const qaEmployee = await prisma.employeeProfile.findUnique({
    where: { authUserId: user.id },
    select: { id: true, role: true, status: true },
  });

  if (
    !qaEmployee ||
    qaEmployee.role !== "QA" ||
    qaEmployee.status !== "ACTIVE"
  ) {
    redirect("/portal/unauthorized");
  }

  const outcome = await prisma.$transaction(async (tx) => {
    const submission = await tx.taskSubmission.findUnique({
      where: { id: submissionId },
      select: {
        id: true,
        taskId: true,
        employeeId: true,
        status: true,
        task: { select: { status: true } },
      },
    });

    if (
      !submission ||
      submission.status !== "PENDING_QA" ||
      submission.task.status !== "PENDING_QA" ||
      submission.employeeId === qaEmployee.id
    ) {
      return false;
    }

    // Claim submission once: only one QA decision can succeed.
    const claimed = await tx.taskSubmission.updateMany({
      where: {
        id: submissionId,
        status: "PENDING_QA",
      },
      data: { status: "QA_IN_PROGRESS" },
    });

    if (claimed.count !== 1) return false;

    // Ensure task is still in the matching state.
    const taskClaimed = await tx.workTask.updateMany({
      where: {
        id: submission.taskId,
        status: "PENDING_QA",
      },
      data: { status: "QA_IN_PROGRESS" },
    });

    if (taskClaimed.count !== 1) {
      throw new Error("Task state changed during QA review.");
    }

    await tx.taskReview.create({
      data: {
        submissionId,
        reviewerId: qaEmployee.id,
        stage: "QA",
        decision,
        feedback: feedback || null,
      },
    });

    const nextStatus =
      decision === "APPROVED" ? "COMPLETED" : "REWORK_REQUIRED";

    await tx.taskSubmission.update({
      where: { id: submissionId },
      data: { status: nextStatus },
    });

    await tx.workTask.update({
      where: { id: submission.taskId },
      data: { status: nextStatus },
    });

    return true;
  });

  if (!outcome) {
    redirect("/portal/qa/queue?error=processed");
  }

  revalidatePath("/portal/qa/queue");
  revalidatePath("/portal/tasker");
  revalidatePath("/portal/reviewer/queue");

  redirect("/portal/qa/queue?result=saved");
}

          )}
        </section>

        <section className="mt-8 rounded-2xl border border-white/10 bg-[#101F39] p-6 sm:p-8">
          <h2 className="text-xl font-bold">Submission History</h2>
          {task.submissions.length === 0 ? (
            <p className="mt-4 text-sm text-slate-300">No submissions yet.</p>
          ) : (
            <div className="mt-6 space-y-4">
              {task.submissions.map((submission) => (
                <article key={submission.id} className="rounded-xl border border-white/10 bg-[#071327] p-5">
                  <div className="flex flex-wrap justify-between gap-2 text-xs text-slate-300">
                    <span>{submission.status.replaceAll("_", " ")}</span>
                    <span>{submission.submittedAt.toLocaleString("en-KE", { timeZone: "Africa/Nairobi" })}</span>
                  </div>

                  <p className="mt-4 whitespace-pre-wrap break-words text-sm leading-7">
                    {submission.content}
                  </p>

                  {submission.reviews.map((review, i) => (
                    <div key={i} className="mt-4 rounded-lg border border-cyan-400/20 p-4">
                      <p className="text-xs font-semibold text-cyan-300">
                        {review.stage}: {review.decision.replaceAll("_", " ")}
                      </p>
                      {review.feedback && (
                        <p className="mt-2 whitespace-pre-wrap text-sm text-slate-300">{review.feedback}</p>
                      )}
                    </div>
                  ))}
                </article>
              ))}
            </div>
          )}
        </section>
      </div>
    </main>
  );
}

"use server";

import { redirect } from "next/navigation";
import { revalidatePath } from "next/cache";
import { requireEmployee } from "@/lib/auth/require-employee";
import { prisma } from "@/lib/db/prisma";

export async function submitTask(formData: FormData): Promise<void> {
  const user = await requireEmployee(["TASKER"]);

  const taskId = String(formData.get("taskId") ?? "");
  const content = String(formData.get("content") ?? "").trim();
  const uuid = /^[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i;

  if (!uuid.test(taskId)) {
    redirect("/portal/tasker");
  }

  const taskUrl = `/portal/tasker/tasks/${taskId}`;

  if (content.length < 20 || content.length > 50000) {
    redirect(`${taskUrl}?error=invalid`);
  }

  const employee = await prisma.employeeProfile.findUnique({
    where: { authUserId: user.id },
    select: { id: true, status: true, role: true },
  });

  if (!employee || employee.status !== "ACTIVE" || employee.role !== "TASKER") {
    redirect("/portal/unauthorized");
  }

  const task = await prisma.workTask.findUnique({
    where: { id: taskId },
    select: { id: true, assignedToId: true, status: true },
  });

  if (!task || task.assignedToId !== employee.id) {
    redirect("/portal/tasker");
  }

  if (!["ASSIGNED", "REWORK_REQUIRED"].includes(task.status)) {
    redirect(`${taskUrl}?error=closed`);
  }

  const outcome = await prisma.$transaction(async (tx) => {
    const changed = await tx.workTask.updateMany({
      where: {
        id: taskId,
        assignedToId: employee.id,
        status: { in: ["ASSIGNED", "REWORK_REQUIRED"] },
      },
      data: { status: "PENDING_REVIEW" },
    });

    if (changed.count !== 1) return false;

    await tx.taskSubmission.create({
      data: {
        taskId,
        employeeId: employee.id,
        content,
        status: "PENDING_REVIEW",
      },
    });

    return true;
  });

  if (!outcome) {
    redirect(`${taskUrl}?error=closed`);
  }

  revalidatePath("/portal/tasker");
  revalidatePath("/portal/reviewer");

  redirect(`${taskUrl}?submitted=1`);
}

export async function signIn(formData: FormData) {
  const email = String(formData.get("email") ?? "").trim();
  const password = String(formData.get("password") ?? "");

  if (!email || !password) {
    redirect("/portal/login?error=missing");
  }

  const supabase = await createClient();

  const { error } = await supabase.auth.signInWithPassword({
    email,
    password,
  });

  if (error) {
    redirect("/portal/login?error=invalid");
  }

  redirect("/portal");
}

export async function signOut() {
  const supabase = await createClient();
  await supabase.auth.signOut();
  redirect("/portal/login");
}

========== lib/auth/require-employee.ts ==========
import "server-only";

import { redirect } from "next/navigation";
import { createClient } from "@/lib/supabase/server";
import { prisma } from "@/lib/db/prisma";

export type PortalRole =
  | "TASKER"
  | "REVIEWER"
  | "QA"
  | "ADMIN"
  | "SUPER_ADMIN";

const ROLES: readonly PortalRole[] = [
  "TASKER",
  "REVIEWER",
  "QA",
  "ADMIN",
  "SUPER_ADMIN",
];

function isPortalRole(value: unknown): value is PortalRole {
  return (
    typeof value === "string" &&
    ROLES.some((role) => role === value)
  );
}

export async function requireEmployee(
  permittedRoles?: readonly PortalRole[]
): Promise<{
  id: string;
  email: string;
  role: PortalRole;
}> {
  const supabase = await createClient();

  const {
    data: { user },
    error,
  } = await supabase.auth.getUser();

  if (error || !user) {
    redirect("/portal/login");
  }

  const role = user.app_metadata?.role;

  if (!isPortalRole(role)) {
    redirect("/portal/unauthorized");
  }

  const employee = await prisma.employeeProfile.findUnique({
    where: { authUserId: user.id },
    select: {
      email: true,
      role: true,
      status: true,
    },
  });

  if (
    !employee ||
    employee.status !== "ACTIVE" ||
    employee.role !== role
  ) {
    redirect("/portal/unauthorized");
  }

  if (user.app_metadata?.must_change_password === true) {
    redirect("/portal/change-password");
  }

  // Super Admin has access to all role-specific portals.
  if (
    permittedRoles &&
    role !== "SUPER_ADMIN" &&
    !permittedRoles.includes(role)
  ) {
    redirect("/portal/unauthorized");
  }

  return {
    id: user.id,
    email: user.email ?? employee.email,
    role,
  };
}

========== lib/supabase/server.ts ==========
import "server-only";
import { createServerClient } from "@supabase/ssr";
import { cookies } from "next/headers";

export async function createClient() {
  const cookieStore = await cookies();

  return createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_PUBLISHABLE_KEY!,
    {
      cookies: {
        getAll() {
          return cookieStore.getAll();
        },
        setAll(items) {
          try {
            items.forEach(({ name, value, options }) => {
              cookieStore.set(name, value, options);
            });
          } catch {
            // Cookie updates in Server Components are handled by Proxy.
          }
        },
      },
    }
  );
}
